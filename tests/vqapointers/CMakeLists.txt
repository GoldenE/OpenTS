set(extracted "")
foreach(source "vqalib/buffer_.cpp" "vqa.cpp")
    set(path "${CMAKE_SOURCE_DIR}/code/${source}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    file(READ "${path}" source_text)
    string(REGEX MATCH "std::intptr_t __cdecl VQA_?Memory_?Handler\\([^\n]+\n\\{[^\n]*(\n[^}][^\n]*)*\n\\}" body "${source_text}")
    if(NOT body)
        message(FATAL_ERROR "Cannot extract VQA memory handler: ${source}")
    endif()
    string(APPEND extracted "${body}\n\n")
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/memory_handlers.inc" CONTENT "${extracted}" @ONLY)
set(audio_path "${CMAKE_SOURCE_DIR}/code/ahandle.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${audio_path}" "${CMAKE_SOURCE_DIR}/code/vqalib/audio.cpp")
file(READ "${audio_path}" audio_handler_source)
string(FIND "${audio_handler_source}" "long __cdecl Open_Audio_Handler(VQAHandleP *vqap, AhandleInitParams *params, long b)" audio_start)
if(audio_start LESS 0)
    message(FATAL_ERROR "Cannot locate production VQA audio-open handler")
endif()
string(SUBSTRING "${audio_handler_source}" ${audio_start} -1 audio_tail)
string(FIND "${audio_tail}" "\t\tVQAConfig *config = &vqap->Config;" audio_end)
if(audio_end LESS 0)
    message(FATAL_ERROR "Cannot locate first accepted VQA audio-open setup statement")
endif()
string(SUBSTRING "${audio_tail}" 0 ${audio_end} audio_prefix)
string(APPEND audio_prefix "\t\treturn VQAERR_NONE;\n\t}\n\treturn VQAERR_AUDIO;\n}\n")
file(READ "${CMAKE_SOURCE_DIR}/code/vqalib/audio.cpp" audio_source)
string(REGEX MATCH "long VQA_OpenAudio\\(VQAHandleP \\*vqap\\)\n\\{[^\n]*(\n[^}][^\n]*)*\n\\}" audio_caller "${audio_source}")
if(NOT audio_caller)
    message(FATAL_ERROR "Cannot extract production VQA audio-open caller")
endif()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/audio_open.inc" CONTENT "${audio_prefix}\n${audio_caller}\n" @ONLY)
add_executable(VQAPointersTest vqa_pointers_test.cpp "${CMAKE_SOURCE_DIR}/code/vqalib/dstream.cpp")
target_compile_features(VQAPointersTest PRIVATE cxx_std_20)
target_include_directories(VQAPointersTest PRIVATE "${CMAKE_SOURCE_DIR}/code" "${CMAKE_SOURCE_DIR}/code/vqalib" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(VQAPointersTest PRIVATE /EHsc $<$<CONFIG:Debug>:/MTd> $<$<CONFIG:Release>:/MT>)
add_test(NAME vqapointers COMMAND VQAPointersTest)

string(FIND "${audio_handler_source}" "namespace {" trace_start)
string(FIND "${audio_handler_source}" "/// use of this is a bug" trace_end)
if(trace_start LESS 0 OR trace_end LESS trace_start)
    message(FATAL_ERROR "Cannot extract bounded VQA audio trace helpers")
endif()
math(EXPR trace_length "${trace_end} - ${trace_start}")
string(SUBSTRING "${audio_handler_source}" ${trace_start} ${trace_length} trace_helpers)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/audio_trace.inc" CONTENT "${trace_helpers}" @ONLY)
add_executable(VQAAudioTraceTest audio_trace_test.cpp)
target_compile_features(VQAAudioTraceTest PRIVATE cxx_std_20)
target_include_directories(VQAAudioTraceTest PRIVATE "${CMAKE_SOURCE_DIR}/code" "${CMAKE_SOURCE_DIR}/code/vqalib" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(VQAAudioTraceTest PRIVATE /EHsc $<$<CONFIG:Debug>:/MTd> $<$<CONFIG:Release>:/MT>)
add_test(NAME vqaaudiotrace COMMAND VQAAudioTraceTest on)
add_test(NAME vqaaudiotrace-disabled COMMAND VQAAudioTraceTest off)
