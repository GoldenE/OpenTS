file(READ "${CMAKE_SOURCE_DIR}/code/version.cpp" version_source)
file(READ "${CMAKE_SOURCE_DIR}/code/winstub.cpp" window_source)
string(APPEND version_source "\n${window_source}")
set(extracted "")
set(functions "VersionClass::VersionClass\\(void\\)[^{]*" "void VersionClass::Init_Clipping\\(void\\)" "unsigned int VersionClass::Clip_Version\\([^)]*\\)" "unsigned int VersionClass::Min_Version\\(void\\)" "unsigned int VersionClass::Max_Version\\(void\\)")
list(APPEND functions "unsigned int Build_Number\\(void\\)")
foreach(function IN LISTS functions)
    string(REGEX MATCH "${function}" signature "${version_source}")
    if(NOT signature)
        message(FATAL_ERROR "Cannot extract architecture version contract: ${function}")
    endif()
    string(FIND "${version_source}" "${signature}" function_start)
    string(SUBSTRING "${version_source}" ${function_start} -1 function_tail)
    string(FIND "${function_tail}" "\n}" function_end)
    if(function_end LESS 0)
        message(FATAL_ERROR "Cannot find architecture version function end")
    endif()
    math(EXPR function_length "${function_end} + 2")
    string(SUBSTRING "${function_tail}" 0 ${function_length} body)
    string(APPEND extracted "${body}\n\n")
endforeach()
foreach(source IN ITEMS version.cpp winstub.cpp saveload.cpp init.cpp)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/code/${source}")
endforeach()
file(READ "${CMAKE_SOURCE_DIR}/code/saveload.cpp" save_source)
string(FIND "${save_source}" "bool Load_Game(const char *file_name)" save_start)
if(save_start LESS 0)
    message(FATAL_ERROR "Cannot locate Load_Game identity guard")
endif()
string(SUBSTRING "${save_source}" ${save_start} -1 save_tail)
string(FIND "${save_tail}" "\tLoadedSaveVersion =" save_end)
if(save_end LESS 0)
    message(FATAL_ERROR "Cannot locate first loaded-state assignment")
endif()
string(SUBSTRING "${save_tail}" 0 ${save_end} save_prefix)
string(FIND "${save_prefix}" "if (info.Get_Internal_Version() != ExpectedGameVersion)" save_guard)
if(save_guard LESS 0)
    message(FATAL_ERROR "Save architecture guard must precede LoadedSaveVersion and session mutation")
endif()
string(APPEND extracted "${save_prefix}\treturn(true);\n}\n\n")
file(READ "${CMAKE_SOURCE_DIR}/code/init.cpp" recording_source)
string(REGEX MATCH "bool Load_Recording_Values\\(CCFileClass & file\\)\n\\{[^}]*\\}" recording_prefix "${recording_source}")
if(NOT recording_prefix)
    message(FATAL_ERROR "Cannot extract early recording identity guard")
endif()
string(APPEND extracted "${recording_prefix}\n\treturn(true);\n}\n")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/architecture_functions.inc" CONTENT "${extracted}" @ONLY)
