set(sources "")
foreach(name cstream lzo1x_c lzo1x_d lzopipe lzostraw pipe straw)
    set(path "${CMAKE_SOURCE_DIR}/code/${name}.cpp")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    file(READ "${path}" source)
    string(REPLACE "#include \"always.h\"" "#include <cassert>\n#include <cstring>\n#include <objidl.h>" source "${source}")
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${name}_production.cpp" CONTENT "${source}" @ONLY)
    list(APPEND sources "${CMAKE_CURRENT_BINARY_DIR}/${name}_production.cpp")
endforeach()

file(READ "${CMAKE_SOURCE_DIR}/code/cstream.h" legacy_header)
string(REPLACE "CStreamClass" "LegacyCStreamClass" legacy_header "${legacy_header}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/legacy_cstream.h" CONTENT "${legacy_header}" @ONLY)
file(READ "${CMAKE_CURRENT_BINARY_DIR}/cstream_production.cpp" legacy_source)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/legacy_read.inc" legacy_read)
string(REGEX MATCH "HRESULT CStreamClass::Read\\(void \\*pv, ULONG cb, ULONG \\*pcbRead\\)\n\\{[^\n]*(\n[^}][^\n]*)*\n\\}" current_read "${legacy_source}")
if(NOT current_read)
    message(FATAL_ERROR "Cannot extract current compressed-stream reader")
endif()
string(REPLACE "${current_read}" "${legacy_read}" legacy_source "${legacy_source}")
string(REPLACE "CStreamClass" "LegacyCStreamClass" legacy_source "${legacy_source}")
string(REPLACE "#include \"cstream.h\"" "#include \"legacy_cstream.h\"" legacy_source "${legacy_source}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/legacy_cstream.cpp" CONTENT "${legacy_source}" @ONLY)
list(APPEND sources "${CMAKE_CURRENT_BINARY_DIR}/legacy_cstream.cpp")
add_executable(LZOStreamsTest lzostreams_test.cpp ${sources})
target_compile_features(LZOStreamsTest PRIVATE cxx_std_20)
target_include_directories(LZOStreamsTest PRIVATE "${CMAKE_SOURCE_DIR}/code" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(LZOStreamsTest PRIVATE NOMINMAX)
target_compile_options(LZOStreamsTest PRIVATE /EHsc /we4302 /we4311 /we4312 $<$<CONFIG:Debug>:/MTd> $<$<CONFIG:Release>:/MT>)
target_link_libraries(LZOStreamsTest PRIVATE ole32 uuid $<$<CONFIG:Debug>:comsuppwd> $<$<CONFIG:Release>:comsuppw>)
add_test(NAME lzostreams COMMAND LZOStreamsTest)
