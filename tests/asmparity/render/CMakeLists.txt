set(path "${CMAKE_SOURCE_DIR}/code/voxlib.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
file(READ "${path}" source_text)
set(extracted "")
foreach(name Pack_Voxel_Position Draw_Voxel_Portable Draw_Voxel_Regular_Normals Draw_Voxel_Reverse_Normals Draw_Voxel_Regular_Normals_Lighting Draw_Voxel_Reverse_Normals_Lighting Draw_Voxel_Regular Draw_Voxel_Reverse)
    string(REGEX MATCH "(static unsigned int|static void|void __cdecl) ${name}\\([^\n]*\\)\n\\{[^\n]*(\n[^}][^\n]*)*\n\\}" body "${source_text}")
    if(NOT body)
        message(FATAL_ERROR "Cannot extract voxel drawer: ${name}")
    endif()
    string(APPEND extracted "${body}\n\n")
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/voxel_functions.inc" CONTENT "${extracted}" @ONLY)

set(path "${CMAKE_SOURCE_DIR}/code/isotype.h")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
file(READ "${path}" source_text)
string(FIND "${source_text}" "#pragma pack(4)" first)
string(FIND "${source_text}" "class IsometricTileTypeClass :" last)
if(first LESS 0 OR last LESS 0)
    message(FATAL_ERROR "Cannot extract TMP layout")
endif()
math(EXPR length "${last} - ${first}")
string(SUBSTRING "${source_text}" ${first} ${length} tmp_layout)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/tmp_layout.inc" CONTENT "${tmp_layout}" @ONLY)

set(path "${CMAKE_SOURCE_DIR}/code/blitblit.h")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
file(READ "${path}" source_text)
set(blitters "")
foreach(name BlitTrans BlitTransXlat BlitTransRemapXlat)
    string(REGEX MATCH "template<class T>\nclass ${name} : public Blitter \\{[^\n]*(\n[^}][^\n]*)*\n\\};" body "${source_text}")
    if(NOT body)
        message(FATAL_ERROR "Cannot extract ${name}")
    endif()
    string(APPEND blitters "${body}\n")
endforeach()
string(REGEX MATCH "template <>\ninline void BlitTransRemapXlat<unsigned short>::BlitForward[^\n]*\n\\{[^\n]*(\n[^}][^\n]*)*\n\\}" body "${source_text}")
if(NOT body)
    message(FATAL_ERROR "Cannot extract retained remap span specialization")
endif()
string(APPEND blitters "${body}\n")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/blitter_templates.inc" CONTENT "${blitters}" @ONLY)
