set(OPENTS_PARITY_DIR "${CMAKE_CURRENT_SOURCE_DIR}")

function(opents_add_asmparity name)
    cmake_parse_arguments(PARITY "" "" "SOURCES;LIBRARIES" ${ARGN})
    set(target "AsmParity${name}")
    add_executable(${target} ${PARITY_SOURCES}
        "${OPENTS_PARITY_DIR}/main.cpp" "${OPENTS_PARITY_DIR}/parity.cpp")
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${OPENTS_PARITY_DIR}" "${CMAKE_SOURCE_DIR}/code" "${OPENTS_GENERATED_DIR}")
    target_compile_options(${target} PRIVATE
        /EHsc /Zc:__cplusplus /fp:precise
        $<$<CONFIG:Debug>:/MTd> $<$<CONFIG:Release>:/MT>
        $<$<EQUAL:${CMAKE_SIZEOF_VOID_P},4>:/arch:SSE2>)
    target_compile_definitions(${target} PRIVATE WIN32 _WINDOWS _MBCS NO_BLOWFISH_DLL)
    if(name STREQUAL "Harness")
        target_compile_definitions(${target} PRIVATE OPENTS_ASM_REFERENCE=1)
    elseif(TARGET OpenTSAsmReference)
        target_compile_definitions(${target} PRIVATE OPENTS_ASM_REFERENCE=1)
        target_link_libraries(${target} PRIVATE OpenTSAsmReference)
    else()
        target_compile_definitions(${target} PRIVATE OPENTS_ASM_REFERENCE=0)
    endif()
    target_link_libraries(${target} PRIVATE ${PARITY_LIBRARIES})
    string(TOLOWER "${name}" suite)
    set(vectors "${OPENTS_PARITY_DIR}/vectors/${suite}.txt")
    set(archive "${OPENTS_PARITY_DIR}/vectors/${suite}.tar.gz")
    if(EXISTS "${archive}")
        set(checksum "${OPENTS_PARITY_DIR}/vectors/${suite}.sha256")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${archive}" "${checksum}")
        set(unpacked "${CMAKE_CURRENT_BINARY_DIR}/vectors/${suite}")
        file(ARCHIVE_EXTRACT INPUT "${archive}" DESTINATION "${unpacked}")
        set(vectors "${unpacked}/${suite}.txt")
        file(READ "${checksum}" expected_sha)
        string(STRIP "${expected_sha}" expected_sha)
        file(SHA256 "${vectors}" actual_sha)
        if(NOT actual_sha STREQUAL expected_sha)
            message(FATAL_ERROR "Decompressed ${suite} vectors differ from their captured SHA256")
        endif()
    endif()
    if(EXISTS "${vectors}")
        add_test(NAME "asmparity-${suite}" COMMAND ${target} --golden "${vectors}")
    elseif(TARGET OpenTSAsmReference)
        add_test(NAME "asmparity-${suite}" COMMAND ${target})
    else()
        message(FATAL_ERROR "Missing persisted assembly vectors for ${suite}: ${vectors}")
    endif()
endfunction()
