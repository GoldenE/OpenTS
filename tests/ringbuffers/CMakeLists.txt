foreach(buffer abuffer zbuffer)
    set(extracted "")
    foreach(extension h cpp)
        set(path "${CMAKE_SOURCE_DIR}/code/${buffer}.${extension}")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
        file(READ "${path}" source)
        string(REGEX REPLACE "#include[^\n]*\n" "" source "${source}")
        string(APPEND extracted "${source}\n")
    endforeach()
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${buffer}_production.inc" CONTENT "${extracted}" @ONLY)
endforeach()
add_executable(RingBuffersTest ringbuffers_test.cpp)
target_compile_features(RingBuffersTest PRIVATE cxx_std_20)
target_include_directories(RingBuffersTest PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(RingBuffersTest PRIVATE /EHsc /we4302 /we4311 /we4312 $<$<CONFIG:Debug>:/MTd> $<$<CONFIG:Release>:/MT>)
add_test(NAME ringbuffers COMMAND RingBuffersTest)
