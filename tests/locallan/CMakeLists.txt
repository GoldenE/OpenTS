set(path "${CMAKE_SOURCE_DIR}/code/wspudp.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
file(READ "${path}" source)
set(extracted "")
foreach(name Send_To Receive_From Open_Socket)
    string(REGEX MATCH "(int|bool) UDPInterfaceClass::${name}[ ]*\\([^\n]*\\)\n\\{[^\n]*(\n[^}][^\n]*)*\n\\}" body "${source}")
    if(NOT body)
        message(FATAL_ERROR "Cannot extract loopback transport function ${name}")
    endif()
    string(APPEND extracted "${body}\n\n")
endforeach()
set(path "${CMAKE_SOURCE_DIR}/code/ipxmgr.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
file(READ "${path}" source)
string(REGEX MATCH "void IPXManagerClass::Configure_LAN\\([^\n]*\\)\n\\{[^\n]*(\n[^}][^\n]*)*\n\\}" body "${source}")
if(NOT body)
    message(FATAL_ERROR "Cannot extract LAN configuration")
endif()
string(APPEND extracted "${body}\n\n")
