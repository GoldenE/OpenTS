set(stbuffer_source "")
foreach(extension h cpp)
    set(path "${CMAKE_SOURCE_DIR}/code/stbuffer.${extension}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    file(READ "${path}" source)
    string(REGEX REPLACE "#include[^\n]*\n" "" source "${source}")
    string(APPEND stbuffer_source "${source}\n")
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/stbuffer_production.inc" CONTENT "${stbuffer_source}" @ONLY)
