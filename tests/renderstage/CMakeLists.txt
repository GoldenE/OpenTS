set(owner_path "${CMAKE_SOURCE_DIR}/code/anim.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${owner_path}")
file(READ "${owner_path}" owner_source)
string(REGEX MATCH "if \\((\\(Loops <= 1 && \\(stage >= Class->Stages[^\n]*)\\) \\{" owner_pingpong "${owner_source}")
if(NOT owner_pingpong)
    message(FATAL_ERROR "Cannot extract animation ping-pong predicate")
endif()
set(pingpong_expression "${CMAKE_MATCH_1}")
string(REGEX MATCH "if \\((\\(Loops <= 1 && stage >= Class->Stages[^\n]*)\\) \\{" owner_loop "${owner_source}")
if(NOT owner_loop)
    message(FATAL_ERROR "Cannot extract animation loop predicate")
endif()
set(loop_expression "${CMAKE_MATCH_1}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/render_owner_predicates.inc" CONTENT "bool Owner_PingPong(int stage, int Loops, OwnerType const * Class) { return ${pingpong_expression}; }\nbool Owner_Loop(int stage, int Loops, OwnerType const * Class) { return ${loop_expression}; }\n" @ONLY)
add_executable(RenderStageTest main.cpp owner.cpp "${CMAKE_SOURCE_DIR}/code/renderstage.cpp" "${CMAKE_SOURCE_DIR}/code/_renderstage.cpp" "${CMAKE_SOURCE_DIR}/code/hdasset.cpp")
target_include_directories(RenderStageTest PRIVATE "${CMAKE_SOURCE_DIR}/code" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(RenderStageTest PRIVATE cxx_std_20)
target_compile_options(RenderStageTest PRIVATE /EHsc)
target_link_libraries(RenderStageTest PRIVATE bcrypt winmm)
add_test(NAME renderstage COMMAND RenderStageTest)
