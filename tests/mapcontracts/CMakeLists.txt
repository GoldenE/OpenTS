set(contract_sources map.cpp display.cpp preview.cpp)
set(extracted "")
foreach(source IN LISTS contract_sources)
    set(path "${CMAKE_SOURCE_DIR}/code/${source}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    file(READ "${path}" source_text)
    if(source STREQUAL "map.cpp")
        set(functions "bool MapClass::In_Radar\\(Cell const & cell\\) const" "void MapClass::Init_Cells\\(void\\)" "bool MapClass::Read_Binary_1\\(Straw & straw\\)" "CellClass \\* MapClass::Iterate\\(void\\)" "void MapClass::Reset_Iterator\\(void\\)")
    elseif(source STREQUAL "display.cpp")
        set(functions "void DisplayClass::Encroach_Shadow\\(void\\)")
    else()
        set(functions "void MapPreviewClass::Blit_Preview\\(HWND window\\)" "bool MapPreviewClass::Read_INI\\(CCINIClass const & ini\\)")
    endif()
    foreach(function IN LISTS functions)
        string(REGEX MATCH "${function}\n\\{[^\n]*(\n[^}][^\n]*)*\n\\}" body "${source_text}")
        if(NOT body)
            message(FATAL_ERROR "Cannot extract map contract function: ${function}")
        endif()
        string(APPEND extracted "${body}\n\n")
    endforeach()
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/map_contract_functions.inc" CONTENT "${extracted}" @ONLY)

add_executable(MapContractsTest map_contracts_test.cpp)
target_compile_features(MapContractsTest PRIVATE cxx_std_20)
target_include_directories(MapContractsTest PRIVATE "${CMAKE_SOURCE_DIR}/code" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(MapContractsTest PRIVATE $<$<CONFIG:Debug>:/MTd /EHsc> $<$<CONFIG:Release>:/MT /EHsc>)
add_test(NAME mapcontracts COMMAND MapContractsTest)
