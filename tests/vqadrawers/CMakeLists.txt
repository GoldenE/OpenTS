function(extract_vqa_function source_file signature output)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source_file}")
    file(READ "${source_file}" source_text)
    string(REGEX MATCH "${signature}[^\n]*\n\\{[^\n]*(\n[^}][^\n]*)*\n\\}" body "${source_text}")
    if(NOT body)
        message(FATAL_ERROR "Cannot extract VQA drawer function ${signature}")
    endif()
    set(${output} "${body}" PARENT_SCOPE)
endfunction()

set(table_source "#include <cassert>\n#include <cstring>\n#include <cstdint>\nextern \"C\" unsigned short * HicolorTable;\n")
foreach(name memset32 UnVQ2_4x4_Table UnVQ2_4x2_Table UnVQ1_4x4_Table UnVQ1_4x2_Table)
    if(name STREQUAL "memset32")
        set(signature "static inline void memset32\\(")
    else()
        set(signature "void __cdecl ${name}\\(")
    endif()
    extract_vqa_function("${CMAKE_SOURCE_DIR}/code/unvqtblc.cpp" "${signature}" body)
    string(APPEND table_source "${body}\n\n")
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/table_drawers.cpp" CONTENT "${table_source}" @ONLY)
